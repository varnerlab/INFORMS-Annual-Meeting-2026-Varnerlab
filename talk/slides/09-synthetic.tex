% Synthetic slide (spec, slide 6): the real-run grammar on the JumpHMM futures, one panel per set, each with its own
% x range (talk/data/syn_axes.csv) and its own SPY and equal-weight lines.
% Title from the data: in both sell-off sets every dot, hedge off and on, sits right of SPY (burst: lowest 0.924
% against 0.865; bear: lowest 0.774 against 0.718). Set sizes from results/synthetic/tables.md, in the one-line
% subtitle. Review round (user, 2026-10-30): the title says median, as each dot is one, and the stress sets are
% defined on the page number's line (design spec §6: of 10,000 market paths ranked by synthetic SPY's drawdown in
% the window, the 200 deepest with a jump burst and the 200 deepest without). With text on that line, the space
% above it is the deck's usual 0.9em, so the panels are 0.27 cm shorter than when the line was empty.
% No legend block (user, 2026-10-30): SPY and Equal run up the left side of their lines at the top of the Burst stress
% panel, whose dots all sit right of both lines, and hedge on and hedge off are stacked just right of the Bear stress
% panel. The dots and range bars are smaller than
% the deck's default, so the 140 dots of each narrow panel bunch less. The panels are 2.95 cm wide, so the hedge key
% ends about as far in from the right edge as the row labels start from the left.
{%
\def\vndotOffRadius{1.4pt}
\def\vndotOnRadius{1.35pt}
\def\vndotBarWidth{1.7pt}
{\renewcommand{\vnfootleft}{Stress sets: of 10,000 paths, the 200 deepest SPY drawdowns with a jump burst, and the 200
  without one.}
\begin{frame}{Simulated sell-offs: every median ends above SPY}
  % a group, not braces: beamer would read a brace group after the title as a frame subtitle
  \begingroup\small 600 typical JumpHMM futures of Jan 2025 – Sep 2026, and 200 in each stress set.\par\endgroup
  \vndotreadaxis{data/syn_axes.csv}{typical}%
  \vndotreadrefs{data/syn_typical_refs.csv}%
  \vndotplot[xmin=\vndotAxisLo, xmax=\vndotAxisHi, xstep=0.1, spy=\vndotSPY, ew=\vndotEW, width=2.95cm,
    height=4.23cm, labelsize=7, title={\vnmedium Typical}]%
    {data/syn_typical_dots.csv}{data/syn_typical_bars.csv}%
  \hspace{0.3cm}%
  \vndotreadaxis{data/syn_axes.csv}{burst_stress}%
  \vndotreadrefs{data/syn_burst_stress_refs.csv}%
  % the shared axis title hangs under all three panels, so it does not count toward this panel's width; it sits 2pt
  % nearer the tick labels than usual
  \pgfplotsset{vndot extra/.style={xlabel style={overlay, yshift=2pt}}}%
  \vndotplot[xmin=\vndotAxisLo, xmax=\vndotAxisHi, xstep=0.1, spy=\vndotSPY, ew=\vndotEW, width=2.95cm,
    height=4.23cm, rowlabels=false, spylabel=SPY, ewlabel=Equal,
    xlabel={terminal wealth ÷ initial wealth, median across futures}, title={\vnmedium Burst stress}]%
    {data/syn_burst_stress_dots.csv}{data/syn_burst_stress_bars.csv}%
  % no space here: the Bear stress panel's first tick label, at its left edge, already opens the same gap as the
  % 0.3cm before the Burst stress panel
  \vndotreadaxis{data/syn_axes.csv}{bear_stress}%
  \vndotreadrefs{data/syn_bear_stress_refs.csv}%
  \pgfplotsset{vndot extra/.style={extra description/.code={\vndothedgekey}}}%
  \vndotplot[xmin=\vndotAxisLo, xmax=\vndotAxisHi, xstep=0.1, spy=\vndotSPY, ew=\vndotEW, width=2.95cm,
    height=4.23cm, rowlabels=false,
    title={\vnmedium Bear stress}]%
    {data/syn_bear_stress_dots.csv}{data/syn_bear_stress_bars.csv}\par
\end{frame}}}

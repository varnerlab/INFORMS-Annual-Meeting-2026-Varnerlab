% Utility slide (spec, slide 2): one frame, two builds. Build 1 is the preference weight and the general problem;
% build 2 keeps the preference weight and replaces the problem with the closed forms under the budget alone.
% House rule: every symbol is defined in the prose before its display.
\begin{frame}{Utility functions turn a preference score into weights}
  % the template sets list text to \normalsize; this frame's lists are smaller so build 1 fits
  \setbeamerfont{itemize/enumerate body}{size=\footnotesize}%
  \small
  \setlength{\abovedisplayskip}{3pt}%
  \setlength{\belowdisplayskip}{4pt}%
  Stock $i$: $n_i$ shares at price $S_i$, weight $w_i = n_i S_i / W$ in a book of wealth $W$. A single-index model of
  daily growth rates gives $\alpha_i$ and $\beta_i$. With market growth signal $\tilde g_M$, market-state signal $\xi$
  from SPY's moving averages and $\beta_{\text{floor}} = 0.1$, the preference weight is
  \[
    \gamma_i = \tanh\!\big((\alpha_i + \beta_i \tilde g_M) \,/\, \max(|\beta_i|, \beta_{\text{floor}})^{\xi}\big).
  \]
  \only<1>{%
    \vspace{-0.6em}
    \begin{columns}[T,onlytextwidth]
      \begin{column}{0.48\textwidth}
        \footnotesize
        Choose $n$ to maximize $U(n; \gamma)$ subject to:
        \begin{itemize}
          \setlength{\itemsep}{0pt}
          \item the budget;
          \item at most 10\% of wealth per name;
          \item volatility capped at SPY's 2014–2024 level;
          \item a turnover budget per rebalance;
          \item hedge on: a cap on the portfolio $\beta$.
        \end{itemize}
      \end{column}
      \begin{column}{0.48\textwidth}
        \footnotesize
        Utilities, summed over $\gamma_i > 0$, with $\eta > 0$ the CES elasticity, $\rho = 1 - 1/\eta$ and $b_i$ a
        minimum holding:
        \begin{itemize}
          \setlength{\itemsep}{0pt}
          \item Cobb-Douglas $\sum \gamma_i \ln n_i$;
          \item CES $\big(\sum \gamma_i n_i^{\rho}\big)^{1/\rho}$;
          \item CES on weights, the same with $w_i$;
          \item Stone–Geary $\sum \gamma_i \ln(n_i - b_i)$.
        \end{itemize}
      \end{column}
    \end{columns}
    \vspace{0.3em}
    {\footnotesize Each solve is a convex cone program: exponential cones for the log utilities, power cones for
      CES.\par}
  }%
  \only<2>{%
    Under the budget alone, with $\eta > 0$ the CES elasticity, the closed forms:
    \begin{itemize}
      \setlength{\itemsep}{0pt}
      \item Cobb-Douglas $w_i \propto \gamma_i$;
      \item CES on weights $w_i \propto \gamma_i^{\eta}$;
      \item CES on shares $n_i \propto (\gamma_i / S_i)^{\eta}$.
    \end{itemize}
    No solver needed, so allocation is instant.\par
  }%
  \vfill
  {\footnotesize\color{vnslidebody} Also compared: minimum variance, the tangency portfolio, and 130/30 long-short
    Cobb-Douglas.\par}
\end{frame}
